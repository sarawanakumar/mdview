Code blocks

One block per supported language (spec 7.1), then blocks that must stay plain (7.4) and blocks that are broken on purpose (7.6). Each language block has keywords, strings, comments and numbers (7.2), and most have a string or comment that spans lines (7.3).

Swift

import Foundation

/* A block comment
   over two lines */
struct Greeter: Sendable {
    let name: String
    func greet(times: Int = 3) -> String {
        let banner = """
            Hello, \(name)!
            """
        return times > 0 ? banner : "" // never empty for 3
    }
}

# Read a config file
def load(path, retries=2):
    """Load the file.

    Tries a few times."""
    for attempt in range(retries):
        if attempt is not None:
            return open(path).read()
    return 'none'
